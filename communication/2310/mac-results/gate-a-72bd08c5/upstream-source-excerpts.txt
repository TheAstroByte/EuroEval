   927	    def determine_available_memory(self) -> int:
   928	        """Return scheduler-visible available cache memory."""
   929	        mode = self._worker.model_runner.scheduler_memory_reporting_mode()
   930
   931	        if mode == "stt_nominal":
   932	            logger.info("STT model: reporting nominal memory for scheduler")
   933	            return STT_SCHED_AVAILABLE_BYTES
   934
   935	        if mode == "paged_attention_capacity":
   936	            overhead = self._worker.model_runner.profile_run()
   937	            self.setup_paged_attention(overhead=overhead)
   938	            backend = self._worker.model_runner.paged_attention_runtime
   939	            if backend is None:
   940	                raise RuntimeError(
   941	                    "Paged attention backend not initialized for capacity reporting"
   942	                )
   943	            block_size_bytes = self._worker.get_cache_block_size_bytes()
   944	            available = backend.num_blocks() * block_size_bytes
   945	            logger.info(
   946	                "Paged attention: reporting MPS cache capacity "
   947	                "(%d blocks × %d bytes = %.2f GB)",
   948	                backend.num_blocks(),
   949	                block_size_bytes,
   950	                available / 1e9,
   951	            )
   952	            return available
   953
   954	        if mode == "paged_attention_layout_budget":
   955	            overhead = self._worker.model_runner.profile_run()
   956	            budget = self.base_kv_budget_bytes(
   957	                metal_limit=self._metal_limit_bytes(),
   958	                fraction=self._memory_fraction(),
   959	                model_memory=self.get_model_memory_usage(),
   960	                overhead=overhead,
   961	            )
   962	            budget -= self._worker.model_runner.draft_scratch_reserve_bytes()
   963	            logger.info(
   964	                "Upstream cache layout: reporting %.2f GB KV budget; "
   965	                "runtime allocation deferred until vLLM KVCacheConfig",
   966	                budget / 1e9,
   967	            )
   968	            return budget

  1043	        logger.info(
  1044	            "Paged attention: using --gpu-memory-utilization=%.2f",
  1045	            fraction,
  1046	        )
